  /** Local, persisted reinforcement. No shared BuildingDef mutation. */
  function raniMaxHp(b: PlacedBuilding): number {
    return buildingHp(b.def) + ((b.raniLevel ?? 1)-1)*80;
  }
  function raniUpgradeCost(b: PlacedBuilding): {gold:number;stone:number}|null {
    if (b.def.paintable || (b.raniLevel ?? 1)>=3) return null;
    const level=b.raniLevel ?? 1;
    return {gold:110*level,stone:6*level};
  }
  function canRaniDemolish(b: PlacedBuilding): boolean {
    if (b.name==='keep') return false;
    if (b.name==='stockpile'||b.name==='granary')
      return state.buildings.filter(x=>x.name===b.name).length>1;
    return true;
  }
  function raniUpgrade(b: PlacedBuilding): void {
    if(!state.buildings.includes(b)) return;
    if(mp){state.notify('대전 중에는 건물을 강화할 수 없습니다.','warn');return;}
    const cost=raniUpgradeCost(b);
    if(!cost){state.notify('최대 강화 단계이거나 강화할 수 없는 시설입니다.','warn');return;}
    if(state.gold<cost.gold||state.stock.stone<cost.stone){
      state.notify('금 '+cost.gold+' · 석재 '+cost.stone+'이 필요합니다.','warn');return;
    }
    state.gold-=cost.gold;state.stock.stone-=cost.stone;
    b.raniLevel=(b.raniLevel??1)+1;
    b.hp=Math.min(raniMaxHp(b),b.hp+80);
    state.notify(b.def.label+' Lv.'+b.raniLevel+' 강화 성공 · 최대 내구도 +80','info');
  }
  const inspector=new RaniBuildingWheel({
    isMatch:!!mp,
    isPresent:b=>state.buildings.includes(b),
    focus:b=>{const [w,d]=b.def.footprint;lookAt(b.x+w/2,b.z+d/2);},
    getPoint:b=>{
      const [w,d]=b.def.footprint,x=b.x+w/2,z=b.z+d/2;
      return iso.worldToScreen(x,terrain.heightAt(x,z),z);
    },
    maxHp:raniMaxHp,upgradeCost:raniUpgradeCost,canDemolish:canRaniDemolish,
    upgrade:raniUpgrade,
    manage:b=>{
      if(b.name==='gatehouse')toggleGate(b);
      else if(b.def.alternate)toggleProduct(b);
    },
    demolish:b=>{
      if(!canRaniDemolish(b))return;
      demolish(b);refreshOverlay(true);
    }
  });
